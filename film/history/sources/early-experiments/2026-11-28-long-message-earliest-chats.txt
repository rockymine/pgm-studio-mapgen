import numpy as np
import networkx as nx
import random
import matplotlib.pyplot as plt

def generate_pvp_map(grid_size, num_islands):
    width, height = grid_size
    grid = np.zeros((width, height))  # Empty grid

    # Step 1: Randomly place islands
    islands = set()
    while len(islands) < num_islands:
        x, y = random.randint(0, width-1), random.randint(0, height-1)
        islands.add((x, y))
    
    island_list = list(islands)

    # Step 2: Build a Minimum Spanning Tree (MST) to connect islands
    G = nx.Graph()
    for i, (x1, y1) in enumerate(island_list):
        for j, (x2, y2) in enumerate(island_list):
            if i != j:
                weight = abs(x1 - x2) + abs(y1 - y2)  # Manhattan distance
                G.add_edge((x1, y1), (x2, y2), weight=weight)

    mst = nx.minimum_spanning_tree(G, weight='weight')

    # Step 3: Visualize the map
    fig, ax = plt.subplots(figsize=(8, 8))
    ax.set_xticks(range(width))
    ax.set_yticks(range(height))
    ax.grid(True, which='both', linestyle='--', linewidth=0.5)

    # Draw islands
    for (x, y) in island_list:
        ax.scatter(x, y, c='red', s=100, label="Island" if 'Island' not in ax.get_legend_handles_labels()[1] else "")

    # Draw paths
    for (x1, y1), (x2, y2) in mst.edges:
        ax.plot([x1, x2], [y1, y2], c='blue', linewidth=2)

    ax.set_xlim(-1, width)
    ax.set_ylim(-1, height)
    ax.set_title("Minecraft PvP Map Layout")
    plt.legend()
    plt.gca().invert_yaxis()  # Invert Y-axis for a more intuitive grid view
    plt.show()

def generate_pvp_map_with_voids(grid_size, num_islands, num_void_separators):
    width, height = grid_size
    grid = np.zeros((width, height))  # Empty grid

    # Step 1: Randomly place islands (red points)
    islands = set()
    while len(islands) < num_islands:
        x, y = random.randint(0, width-1), random.randint(0, height-1)
        islands.add((x, y))
    
    island_list = list(islands)

    # Step 2: Build a Minimum Spanning Tree (MST) to connect islands
    G = nx.Graph()
    for i, (x1, y1) in enumerate(island_list):
        for j, (x2, y2) in enumerate(island_list):
            if i != j:
                weight = abs(x1 - x2) + abs(y1 - y2)  # Manhattan distance
                G.add_edge((x1, y1), (x2, y2), weight=weight)

    mst = nx.minimum_spanning_tree(G, weight='weight')

    # Step 3: Generate void separators (black points) and intersecting void lines (green)
    void_separators = set()
    while len(void_separators) < num_void_separators:
        x, y = random.randint(0, width-1), random.randint(0, height-1)
        void_separators.add((x, y))
    
    void_list = list(void_separators)

    void_lines = []
    for i in range(0, len(void_list) - 1, 2):  # Pair points to create void lines
        void_lines.append((void_list[i], void_list[i+1]))

    # Step 4: Identify intersections between MST paths and void lines
    void_intersections = set()
    for (x1, y1), (x2, y2) in mst.edges:
        for (vx1, vy1), (vx2, vy2) in void_lines:
            # Check if any MST path crosses the void line (simple bounding box check)
            if min(vx1, vx2) <= max(x1, x2) and min(vy1, vy2) <= max(y1, y2) and \
               max(vx1, vx2) >= min(x1, x2) and max(vy1, vy2) >= min(y1, y2):
                void_intersections.add(((x1, y1), (x2, y2)))

    # Remove MST edges at intersections
    for edge in void_intersections:
        if edge in mst.edges:
            mst.remove_edge(*edge)

    # Step 5: Visualize the updated map
    fig, ax = plt.subplots(figsize=(8, 8))
    ax.set_xticks(range(width))
    ax.set_yticks(range(height))
    ax.grid(True, which='both', linestyle='--', linewidth=0.5)

    # Draw islands (red dots)
    for (x, y) in island_list:
        ax.scatter(x, y, c='red', s=100, label="Island" if 'Island' not in ax.get_legend_handles_labels()[1] else "")

    ax.set_xlim(-1, width)

        node_mapping = {}
        for node in source_nodes:
            mirrored_x = self.width - 1 - node.position[0]
            mirrored_pos = (mirrored_x, node.position[1])
            mirrored_id = node.id.replace(f"_{source_team}", f"_{target_team}")
            
            mirrored_node = GameNode(mirrored_id, node.node_type, mirrored_pos, target_team)
            self.nodes[mirrored_id] = mirrored_node
            node_mapping[node.id] = mirrored_id
        
        # Create mirrored edges
        for edge in source_edges:
            mirrored_start = node_mapping[edge.start_node]
            mirrored_end = node_mapping[edge.end_node]
            
            # Mirror waypoints
            mirrored_waypoints = []
            for x, y in edge.waypoints:
                mirrored_x = self.width - 1 - x
                mirrored_waypoints.append((mirrored_x, y))
            
            self._create_lane(mirrored_start, mirrored_end, mirrored_waypoints)
    
    def _create_void_connections(self):
        """Create connections across the void between teams"""
        team0_void = [n for n in self.nodes.values() if n.node_type == NodeType.VOID_ACCESS and n.team == 0]
        team1_void = [n for n in self.nodes.values() if n.node_type == NodeType.VOID_ACCESS and n.team == 1]
        
        # Connect corresponding void access points
        for i in range(min(len(team0_void), len(team1_void))):
            void0 = team0_void[i]
            void1 = team1_void[i]
            
            void_edge_id = f"void_connection_{i}"
            waypoints = [void0.position, void1.position]
            void_edge = GameEdge(void_edge_id, EdgeType.VOID_CONNECTION, void0.id, void1.id, waypoints)
            self.edges[void_edge_id] = void_edge
            
            # Add to graph
            self.graph.add_edge(void0.id, void1.id, weight=self.void_width)

Overcast offers a variety of Gamemodes for which Builders like myself create maps. One gamemode I particularly like is Capture the Wool, for short CTW. In this gamemode two or more teams try to be the first to capture the enemies wool or wools. Wools are usually located in wool rooms, structures that contain a spawner that spawns wool blocks of a specific color. This structure must be defended by each team to prevent the opponents from getting into it. When an enemy player ultimately gets the wool and is able to escape they must navigate the map to place the wool on a designated block, the wool monument. The plugin recognizes this and marks the wool as captured. Once a team has captured all the enemies wools they win. Maps are generally built inside the void so that players can fall or get knocked out of the world. When a player dies they respawn at the spawn and automatically get reequipped with tools that are defined in the spawn-kit. This usually consists of basic armor, tools, some blocks, sword and bow and some food for healing. Outside of the spawn structure you will sometimes find iron blocks to craft better armor. The wools may be located to the left and right of spawn on seperate lanes that can be thinner or wider. To the sides of the lanes you will find void gaps where players cannot place blocks. This way defenders are able to dig out the lane and build a defense that the offending team must pass. In the early stages players will rush the wool with no armor. Small walls might be set up on the lanes to slow down players. When a bigger defense has been dug many games end with skybridges. Players built a staircase until they reach the maximum height allowed to build, which is defined by the builders, and then continuously expand it with blocks until they reach the wall and jump down with water buckets or ride a boat into the wool. The opposing teams sides are also usually seperated by void and can contain small islands. In general, the layout is symmetric.
